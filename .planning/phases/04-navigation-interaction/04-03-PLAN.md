      ```typescript
      const handleWarpToArtist = () => {
        const state = useAppStore.getState()
        const artistStars = state.stars.filter(s => s.artist === star.artist)
        if (artistStars.length === 0) return
        const centroid: [number, number, number] = [0, 0, 0]
        for (const s of artistStars) {
          centroid[0] += s.position[0]
          centroid[1] += s.position[1]
          centroid[2] += s.position[2]
        }
        centroid[0] /= artistStars.length
        centroid[1] /= artistStars.length
        centroid[2] /= artistStars.length
        state.setSelectedStar(null)
        state.startWarp({ position: centroid })
      }
      ```

    **Step 2: Create StarClickHandler.tsx (proximity picking + warp trigger)**

    Create `src/components/canvas/StarClickHandler.tsx`. This is an R3F component mounted inside Canvas that listens for click events and uses proximity-based picking (per D-06, RESEARCH.md Pattern 3).

    ```typescript
    "use client"
    import { useCallback, useEffect } from 'react'
    import { useThree } from '@react-three/fiber'
    import * as THREE from 'three'
    import { useAppStore } from '@/lib/store'
    import { findNearestStar } from '@/lib/star-picking'
    ```

    Module-scope temp vector:
    ```typescript
    const _tempVec3 = new THREE.Vector3()
    ```

    The component:
    - Gets `camera`, `gl`, `size` from `useThree()`
    - Creates a `projectFn` callback that projects a star's world position to NDC:
      ```typescript
      const projectFn = useCallback((position: [number, number, number]) => {
        _tempVec3.set(position[0], position[1], position[2])
        _tempVec3.project(camera as THREE.PerspectiveCamera)
        return { x: _tempVec3.x, y: _tempVec3.y, z: _tempVec3.z }
      }, [camera])
      ```
    - Registers a `click` event listener on `gl.domElement`:
      ```typescript
      const handleClick = (e: MouseEvent) => {
        // Only handle left click
        if (e.button !== 0) return

        const state = useAppStore.getState()

        // Don't pick stars during warp or search
        if (state.cameraMode === 'warping' || state.searchOpen) return

        // Convert mouse position to NDC
        const ndcX = (e.clientX / size.width) * 2 - 1
        const ndcY = -(e.clientY / size.height) * 2 + 1

        const star = findNearestStar(ndcX, ndcY, state.stars, projectFn)

        if (star) {
          // Star found: warp to it (per D-06)
          state.startWarp({ position: star.position, starId: star.id })
        } else if (state.selectedStar) {
          // Clicked empty space while card is open: dismiss (per UI-SPEC)
          state.setSelectedStar(null)
          state.setCameraMode('cruising')
        }
      }
      ```
    - Also registers an Escape key handler on `window`:
      ```typescript
      const handleKeyDown = (e: KeyboardEvent) => {
        if (e.code === 'Escape') {
          const state = useAppStore.getState()
          if (state.searchOpen) {
            state.setSearchOpen(false)
            return
          }
          if (state.selectedStar) {
            state.setSelectedStar(null)
            state.setCameraMode('cruising')
          }
        }
      }
      ```
    - Returns `null` (no JSX).
